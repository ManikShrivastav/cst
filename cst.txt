Program 1A — Stability using closed-loop poles
clc;
clear;
close all;
K = 10;
G = tf(K,[1 6 8 0]);
T = feedback(G,1);
poles = pole(T)
if all(real(poles)<0)
disp('System is Stable');
elseif any(real(poles)>0)
disp('System is Unstable');
else
disp('System is Marginally Stable');
end

Program 7B — LQR with step response and initial condition
clc;
clear;
close all;
% System matrices
A = [0 1;-1 -0.2];
B = [0;1];
C = [1 0];
D = 0;
% Weighting matrices
Q = [10 0;0 1];
R = 1;
% LQR gain
[K,S,e] = lqr(A,B,Q,R);
disp('Optimal LQR Gain K:');
disp(K);
disp('Closed Loop Poles:');
disp(e);

% Closed loop system
A_cl = A - B*K;
sys_cl = ss(A_cl,B,C,D);
% Step response
step(sys_cl);
grid on;
% Initial condition
x0 = [2;0];
t = 0:0.01:10;
[y,t_out,x] = initial(sys_cl,x0,t);
% Plot results
figure;
plot(t_out,x,'LineWidth',1.5);
grid on;
title('LQR Controlled System Response from Initial State');
xlabel('Time (seconds)');
ylabel('State Values');
legend('Position (x1)','Velocity (x2)');
